Install Ollama and download the models

Ollama runs LLMs on your own computer. It works as a small server in the background, and we send it questions from the terminal or from Python. Pick your system at the top, then install it:

Windows · cmd / PowerShell
winget install --id Ollama.Ollama -e

When it finishes, close the terminal and open a new one. Ollama starts by itself in the system tray (the llama icon). No winget? Download the installer from ollama.com/download and run it.

macOS · Homebrew
brew install ollama
brew services start ollama

The second line starts the Ollama server in the background and keeps it starting at every login. No Homebrew? Download the app from ollama.com/download, move it to Applications and open it once.

Linux
curl -fsSL https://ollama.com/install.sh | sh

The script installs Ollama and registers it as a service, so it is already running.

Create first_chat.py. From the lecture's five parts of a prompt, the role and context go in the system message and the task goes in the user message:

first_chat.py
import sys
import ollama

MODEL = sys.argv[1] if len(sys.argv) > 1 else "gemma3:4b"

messages = [
    # Role + context: who the model is and what it can't know by itself
    {"role": "system", "content": (
        "You are an assistant for the facility team of a university engineering building. "
        "Answer in Korean, in at most 3 short sentences. If you are not sure, say so."
    )},
    # Task
    {"role": "user", "content": "3층 화장실 천장에서 물이 새요. 시설팀은 먼저 무엇을 해야 하나요?"},
]

stream = ollama.chat(model=MODEL, messages=messages, stream=True)
for chunk in stream:
    print(chunk.message.content, end="", flush=True)
    if chunk.done:
        last = chunk

print("\n")
print(f"Model          : {MODEL}")
print(f"Prompt tokens  : {last.prompt_eval_count}")
print(f"Answer tokens  : {last.eval_count}")
print(f"Speed          : {last.eval_count / (last.eval_duration / 1e9):.1f} tokens/s")

Complaint triage: free text in, JSON out

Complaints arrive as free text, but a work-order system needs fixed fields: category, priority, location, trade. We define the fields with pydantic and pass them as format=; Ollama then forces the model to write only JSON of that shape.

triage.py
import re
from typing import Literal

import matplotlib.pyplot as plt
import ollama
import pandas as pd
from pydantic import BaseModel

MODEL = "gemma3:4b"


# 1. The shape of the answer we want back (the "Format" part of the prompt)
class Triage(BaseModel):
    category: Literal["plumbing", "hvac", "electrical", "cleaning", "safety", "elevator", "other"]
    priority: Literal["urgent", "high", "normal", "low"]
    location: str        # building / floor / room, as written in the complaint
    trade: str           # who should go: plumber, electrician, ...
    summary_en: str      # one short English sentence


SYSTEM = """You triage facility complaints for a university engineering building.
Read the complaint (in Korean) and fill in every field.
category guide:
- plumbing: water leaks, pipes, toilets, drains, water dispensers
- hvac: air conditioning, heating, ventilation, air quality, air purifiers
- electrical: lights, sockets, power, projectors, burning smell from wiring
- cleaning: cleaning, rubbish, mould, toilet paper and other supplies
- safety: fire extinguishers, exit signs, blocked escape routes, things that may fall
- elevator: lifts and their buttons
- other: anything else (doors, windows, paint, parking)
priority rules:
- urgent: people may get hurt now, someone is trapped, fire/smoke/burning smell, water pouring
- high: a room cannot be used, a safety device is missing or off, a class is affected
- normal: needs fixing this week
- low: cosmetic or supplies
Example: "2층 복도 전등 하나가 나갔어요" -> category electrical, priority normal, location "2층 복도", trade electrician."""


def mask_phone(text: str) -> str:
    """Privacy first: hide phone numbers before the text goes to any model."""
    return re.sub(r"01[0-9]-?\d{3,4}-?\d{4}", "010-****-****", text)


def triage(text: str) -> Triage:
    for temperature in (0, 0.3):             # 0 = same input, same answer; 0.3 = second try
        try:
            r = ollama.chat(
                model=MODEL,
                messages=[{"role": "system", "content": SYSTEM},
                          {"role": "user", "content": mask_phone(text)}],
                format=Triage.model_json_schema(),   # force valid JSON in this shape
                options={"temperature": temperature, "repeat_penalty": 1.1, "num_predict": 300},
            )
            return Triage.model_validate_json(r.message.content)
        except Exception as e:               # small models sometimes break the JSON
            print(f"   retrying ({e.__class__.__name__})")
    raise RuntimeError("model failed twice on: " + text)


# 2. Triage every complaint
df = pd.read_csv("data/complaints_ko.csv")
rows = []
for _, c in df.iterrows():
    t = triage(c.text)
    print(f"{c.id}  {t.category:<10} {t.priority:<7} {t.location:<16} {t.summary_en}")
    rows.append(t.model_dump())
out = pd.concat([df, pd.DataFrame(rows)], axis=1)
out.to_csv("triage_results.csv", index=False, encoding="utf-8-sig")

# 3. How often did the model agree with the facility team's labels?
cat_acc = (out.category == out.true_category).mean()
pri_acc = (out.priority == out.true_priority).mean()
print(f"\nCategory accuracy: {cat_acc:.0%}   Priority accuracy: {pri_acc:.0%}")
print("\nDisagreements:")
print(out[(out.category != out.true_category) | (out.priority != out.true_priority)]
      [["id", "category", "true_category", "priority", "true_priority"]].to_string(index=False))

# 4. Count complaints per category
counts = out.category.value_counts()
ax = counts.plot.barh(color="#0e7b7b", figsize=(7, 4))
ax.invert_yaxis()
ax.set_xlabel("complaints")
ax.set_title("Complaints by category (model triage)")
plt.tight_layout()
plt.savefig("triage_by_category.png", dpi=150)
print("\nSaved triage_results.csv and triage_by_category.png")

Ask the manual: build a mini RAG

The model has never seen our building's manual. Ask it anyway and it will make up a plausible answer (hallucination). RAG first finds the parts of the manual related to the question, pastes them into the prompt, and says "answer only from these". Four steps:

1 · chunkCut the PDF text into small chunks of 4 lines and remember each chunk's page.
2 · embedTurn each chunk into a vector with bge-m3. Similar meaning, similar vectors.
3 · searchTurn the question into a vector too and find the 3 most similar chunks (cosine similarity).
4 · answerHave the model answer only from those chunks, citing the page like [p.4].
ask_manual.py
import sys

import numpy as np
import ollama
from pypdf import PdfReader

CHAT_MODEL = "gemma3:4b"
EMBED_MODEL = "bge-m3"          # turns text into vectors; works well for Korean
PDF = "data/ahu2_manual_ko.pdf"

# 1. Read the manual and cut it into small chunks, remembering the page of each
chunks = []                      # list of (page, text)
for page_no, page in enumerate(PdfReader(PDF).pages, start=1):
    lines = [ln.strip() for ln in page.extract_text().splitlines() if ln.strip()]
    for i in range(0, len(lines), 4):                 # 4 lines per chunk
        chunks.append((page_no, " ".join(lines[i:i + 4])))
print(f"{len(chunks)} chunks from {PDF}")

# 2. Embed every chunk once: text -> vector of numbers
vectors = np.array(ollama.embed(model=EMBED_MODEL, input=[t for _, t in chunks]).embeddings)
vectors /= np.linalg.norm(vectors, axis=1, keepdims=True)


def search(question: str, k: int = 3):
    q = np.array(ollama.embed(model=EMBED_MODEL, input=question).embeddings[0])
    scores = vectors @ (q / np.linalg.norm(q))        # cosine similarity
    best = np.argsort(scores)[::-1][:k]
    return [(chunks[i][0], chunks[i][1], scores[i]) for i in best]


def ask(question: str) -> str:
    hits = search(question)
    print("\nRetrieved:")
    for page, text, score in hits:
        print(f"  p.{page}  score {score:.2f}  {text[:70]}...")
    sources = "\n".join(f"[p.{page}] {text}" for page, text, _ in hits)
    r = ollama.chat(model=CHAT_MODEL, options={"temperature": 0}, messages=[
        {"role": "system", "content": (
            "Answer the facility technician's question in Korean using ONLY the sources. "
            "Cite the page like [p.4] after each fact. "
            "If the sources do not contain the answer, reply exactly: 매뉴얼에 없는 내용입니다.")},
        {"role": "user", "content": f"Sources:\n{sources}\n\nQuestion: {question}"},
    ])
    return r.message.content


if __name__ == "__main__":
    question = sys.argv[1] if len(sys.argv) > 1 else "AHU-2 프리필터는 언제 교체해야 하나요?"
    print(f"\nQ: {question}")
    print("A:", ask(question))

Photo inspection: a vision model drafts the inspection report

gemma3:4b can read photos too. We send it the Week 4 corridor photo, get the safety and housekeeping findings back as a JSON list, and turn them into a markdown report that a person reviews and signs.

inspect_photo.py
import sys
from datetime import date
from typing import Literal

import ollama
from pydantic import BaseModel, Field

MODEL = "gemma3:4b"             # gemma3 can read images as well as text
PHOTO = sys.argv[1] if len(sys.argv) > 1 else "data/site_photo_corridor.jpg"


class Finding(BaseModel):
    item: str = Field(description="항목, 한국어 (예: 소화기)")
    where: str = Field(description="사진 속 위치, 한국어 (예: 왼쪽 선반 아래)")
    issue: str = Field(description="문제 내용 한 문장, 한국어. 문제가 없으면 '이상 없음'")
    severity: Literal["high", "medium", "low", "ok"]


class Inspection(BaseModel):
    findings: list[Finding]


PROMPT = """당신은 대학 건물 시설팀의 순회점검 담당자입니다.
이 복도 사진을 보고 안전·관리 문제를 3~6개 찾으세요:
피난통로 적치물, 걸려 넘어질 위험, 폐기물, 소화기, 유도등, 파손, 누수.
제자리에 있고 문제없는 안전 설비도 severity "ok" 로 적으세요.
사진에서 실제로 보이는 것만 적고, 모든 내용을 한국어로 쓰세요."""

r = ollama.chat(
    model=MODEL,
    messages=[{"role": "user", "content": PROMPT, "images": [PHOTO]}],
    format=Inspection.model_json_schema(),
    options={"temperature": 0, "repeat_penalty": 1.1},
)
result = Inspection.model_validate_json(r.message.content)

# Write a short inspection report a person must review and sign
order = {"high": 0, "medium": 1, "low": 2, "ok": 3}
lines = [f"# 순회점검 보고서 (초안) — {date.today()}", "",
         f"사진: `{PHOTO}` · 작성: {MODEL} (AI 초안)", "",
         "| # | 항목 | 위치 | 내용 | 심각도 |", "|---|---|---|---|---|"]
for i, f in enumerate(sorted(result.findings, key=lambda f: order[f.severity]), start=1):
    lines.append(f"| {i} | {f.item} | {f.where} | {f.issue} | {f.severity} |")
lines += ["", "검토자 (Reviewed by): ____________   서명: ________", "",
          "> AI가 작성한 초안입니다. 현장 확인 후 승인하세요."]
report = "\n".join(lines)
open("inspection_report.md", "w", encoding="utf-8").write(report)
print(report)
print("\nSaved inspection_report.md")

Mini agent: an LLM that uses tools and asks for approval

Each turn, the agent picks one next step: read a sensor, search the manual, create a work order, or finish. Python runs that tool and tells the model the result (plan → act → observe → repeat). create_work_order changes the real world, so it only runs when a person types y.

It reuses the search function from step 3's ask_manual.py, so keep both files in the same folder. fm_agent.py:

fm_agent.py
import csv
import sys
from datetime import datetime
from typing import Literal

import ollama
from pydantic import BaseModel

from ask_manual import search       # the retrieval function from step 3

MODEL = "gemma3:4b"

# ---- 1. The tools: plain Python functions the model may ask us to run ----
SENSORS = {   # pretend BAS readings (in real life: an API or database query)
    "502호": {"temp_c": 31.5, "setpoint_c": 24, "served_by": "AHU-2", "ahu_alarm": "E01"},
    "301호": {"temp_c": 23.8, "setpoint_c": 24, "served_by": "AHU-2", "ahu_alarm": None},
    "407호": {"temp_c": 24.6, "setpoint_c": 24, "served_by": "AHU-2", "ahu_alarm": None},
}


def get_sensor(room: str) -> str:
    return str(SENSORS.get(room.strip(), "no sensor for that room"))


def search_manual(query: str) -> str:
    return "\n".join(f"[p.{p}] {t}" for p, t, _ in search(query, k=3))


def create_work_order(text: str) -> str:
    # This tool changes the real world, so a person must approve it first
    print(f"\n  >>> The agent wants to create a work order:\n      {text}")
    if input("  >>> Approve? (y/n) ").strip().lower() != "y":
        return "REJECTED by the duty manager. Do not create it again."
    with open("work_orders.csv", "a", newline="", encoding="utf-8-sig") as f:
        csv.writer(f).writerow([datetime.now().isoformat(timespec="minutes"), text])
    return "work order created"


TOOLS = {"get_sensor": get_sensor, "search_manual": search_manual, "create_work_order": create_work_order}


# ---- 2. Each turn the model answers with ONE next step in this shape ----
class Step(BaseModel):
    thought: str     # why, in one sentence
    tool: Literal["get_sensor", "search_manual", "create_work_order", "finish"]
    argument: str    # room / search text / work-order text / final answer to the user


SYSTEM = """You are a facility-management agent for a university building. Work step by step.
Tools:
- get_sensor(argument = room, e.g. "502호"): current temperature, setpoint, AHU and its alarm code
- search_manual(argument = short Korean search text, e.g. "E01 알람 조치"): searches the AHU-2 O&M manual
- create_work_order(argument = Korean work-order text: location, problem, action from the manual)
- finish(argument = final answer to the user, in Korean)
Check the sensor first. If there is an alarm code, look up what to do in the manual before acting,
and put the manual's action and page in the work order.
Use one tool per step. Never invent readings or manual content."""

task = sys.argv[1] if len(sys.argv) > 1 else "502호 강의실이 너무 덥다는 민원이 왔어. 원인을 확인하고 필요하면 작업지시를 올려 줘."
messages = [{"role": "system", "content": SYSTEM}, {"role": "user", "content": task}]
print(f"TASK: {task}")

# ---- 3. The agent loop: plan -> act -> observe -> repeat ----
for turn in range(1, 7):
    r = ollama.chat(model=MODEL, messages=messages, format=Step.model_json_schema(),
                    options={"temperature": 0, "repeat_penalty": 1.1})
    step = Step.model_validate_json(r.message.content)
    print(f"\n[{turn}] {step.tool}({step.argument!r})\n    thought: {step.thought}")
    if step.tool == "finish":
        print(f"\nANSWER: {step.argument}")
        break
    observation = TOOLS[step.tool](step.argument)
    print(f"    result : {observation[:160]}")
    messages.append({"role": "assistant", "content": r.message.content})
    messages.append({"role": "user", "content": f"Result of {step.tool}: {observation}"})
else:
    print("\nStopped after 6 steps without finishing.")
